5 · Named attachment frame

Expose a named local AnchorFrame from a component when the datum belongs to that reusable part.

Compiled: 17 native commandsSchematic illustration only; run the native CI job for a SolidCraft render.

Python source

class Arm:
    def anchors(self):
        return {
            "tip": AnchorFrame(
                origin=(20, 0, 5),
                z=RIGHT,
                x=UP,
            ),
        }

    def build(self):
        return Box(40, 10, 10)


class Marker:
    def anchors(self):
        return {
            "mount": AnchorFrame(
                origin=(0, 0, 5),
                z=LEFT,
                x=UP,
            ),
        }

    def build(self):
        return Box(10, 10, 10)


class CustomAnchor:
    def build(self):
        arm = Arm().rotate(Axis.Z, 90)
        marker = Marker().attach(
            arm,
            parent="tip",
            child="mount",
        )
        return arm + marker
