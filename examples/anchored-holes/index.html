2 · Two holes

An extruded rectangle and two patterned through-holes.

Compiled: 13 native commandsNative SolidCraft 0.30.1 render, verified against this source commit.

Python source

@dataclass
class AnchoredHoles:
    """Two through holes, each 20 mm from the center along the Y axis."""

    width: float = 100
    depth: float = 60
    thickness: float = 10
    spacing: float = 40

    def build(self):
        plate = Rectangle2D(self.width, self.depth).extrude(self.thickness)
        holes = Hole(d=8, depth=Depth.THROUGH_ALL).pattern(
            count=2, spacing=self.spacing, axis=Axis.Y
        ).on(plate, face=Faces.TOP)
        plate + holes
